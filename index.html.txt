    <!-- Custom Physics & Interaction Logic -->
    <script>
      // Kinematic Constants
      const g = 9.8;               // Gravity acceleration (m/s^2)
      const launchSpeed = 5.0;     // Initial velocity of the bullet (m/s) - Change this value to test variations!

      // Fetch 3D Elements from DOM
      const monkey = document.getElementById('monkey');
      const bullet = document.getElementById('bullet');
      const aimLine = document.getElementById('aim-line');

      // Coordinate Profiles (X, Y, Z)
      const hPos = { x: -1, y: 0, z: -3 };
      const mPos = { x: 1, y: 1.5, z: -3 };

      let isPlaying = false;
      let startTime = 0;

      // Listen for a screen tap anywhere on the iPad
      window.addEventListener('click', () => {
        if (isPlaying) {
          // If the simulation is running, tap again to reset
          resetExperiment();
          return;
        }
        startExperiment();
      });

      function startExperiment() {
        isPlaying = true;
        startTime = performance.now();
        bullet.setAttribute('visible', 'true');
        aimLine.setAttribute('visible', 'false'); // Hide aiming line upon launch

        // Calculate 3D Directional Vector from Hunter to Monkey
        const dx = mPos.x - hPos.x;
        const dy = mPos.y - hPos.y;
        const dz = mPos.z - hPos.z;
        const distance = Math.sqrt(dx*dx + dy*dy + dz*dz);

        // Break down initial velocity into X, Y, and Z vector components (V0x, V0y, V0z)
        const vx = (dx / distance) * launchSpeed;
        const vy = (dy / distance) * launchSpeed;
        const vz = (dz / distance) * launchSpeed;

        // Frame-by-frame physics loops for smooth tracking animations
        function updatePhysics() {
          if (!isPlaying) return;

          // Calculate time passed since firing in seconds (t)
          const t = (performance.now() - startTime) / 1000;

          // [Formula 1] Monkey Free Fall: Y = Y0 - 0.5 * g * t^2
          const monkeyY = mPos.y - (0.5 * g * t * t);
          
          // [Formula 2] Bullet Projectile Motion:
          // X & Z: Constant linear horizontal speed (X = X0 + Vx * t)
          // Y: Constant vertical deceleration due to gravity (Y = Y0 + Vy * t - 0.5 * g * t^2)
          const bulletX = hPos.x + (vx * t);
          const bulletY = hPos.y + (vy * t) - (0.5 * g * t * t);
          const bulletZ = hPos.z + (vz * t);

          // Update the positions of elements in 3D AR space
          monkey.setAttribute('position', `${mPos.x} ${monkeyY} ${mPos.z}`);
          bullet.setAttribute('position', `${bulletX} ${bulletY} ${bulletZ}`);

          // Impact Collision Detection: When the bullet reaches the monkey's horizontal grid line
          if (bulletX >= mPos.x) {
            isPlaying = false;
            // Freeze positions immediately and trigger alert explaining the proof
            alert(`DIRECT HIT!\nBoth objects intersected perfectly in mid-air.\nNotice how their Y-coordinates (height) are identical at impact, proving gravity drops them at the exact same rate!`);
            return;
          }

          // Boundary fallback: Stop if objects fall below a virtual floor limit
          if (monkeyY < -2 || bulletY < -2) {
            isPlaying = false;
            return;
          }

          requestAnimationFrame(updatePhysics);
        }

        requestAnimationFrame(updatePhysics);
      }

      function resetExperiment() {
        isPlaying = false;
        // Snap objects back to initial vectors
        monkey.setAttribute('position', `${mPos.x} ${mPos.y} ${mPos.z}`);
        bullet.setAttribute('position', `${hPos.x} ${hPos.y} ${hPos.z}`);
        bullet.setAttribute('visible', 'false');
        aimLine.setAttribute('visible', 'true');
      }
    </script>
